  function findControlBar(video) {
    const root = video.closest('.video-js') || video.closest('[class*="player" i]') || video.parentElement;
    if (!root) return null;
    const sels = ['.vjs-control-bar', '[class*="control-bar" i]', '[class*="controls" i]'];
    for (const s of sels) {
      const el = root.querySelector(s);
      if (el && !el.closest('#wp-container') && el !== kickBtn) return el;
    }
    return null;
  }

  function mountInBar(bar) {
    let first = null;
    ANCHORS.forEach(sel => {
      bar.querySelectorAll(sel).forEach(el => {
        if (el === kickBtn || el.contains(kickBtn)) return;
        if (!first || (first.compareDocumentPosition(el) & Node.DOCUMENT_POSITION_PRECEDING)) first = el;
      });
    });
    kickBtn.classList.remove('cs-float');
    if (first && first.parentNode) first.parentNode.insertBefore(kickBtn, first);
    else bar.appendChild(kickBtn);
    kickBtnHost = bar;
  }

  function mountFloating(root) {
    try { if (getComputedStyle(root).position === 'static') root.style.position = 'relative'; } catch (e) {}
    kickBtn.classList.add('cs-float');
    root.appendChild(kickBtn);
    kickBtnHost = root;
  }

  function removeKickBtn() {
    if (kickBtn && kickBtn.parentNode) kickBtn.parentNode.removeChild(kickBtn);
    kickBtnHost = null;
  }

  function findHeaderAnchor() {
    const cands = Array.from(document.querySelectorAll('button, a')).filter(el => {
      if (el === kickBtn || el.closest('#wp-container') || el.closest('.video-js')) return false;
      const r = el.getBoundingClientRect();
      return r.width > 0 && r.top >= 0 && r.bottom < 130 && r.width < 90 && r.height < 90 && el.querySelector('svg,img');
    });
    if (cands.length < 2) return null;
    const search = cands.find(el => /search|pesquis|busc/i.test(el.getAttribute('aria-label') || el.title || ''));
    if (search) return search;
    cands.sort((a, b) => a.getBoundingClientRect().left - b.getBoundingClientRect().left);
    return cands[cands.length - 2];
  }

  function ensureKickButton() {
    if (session) { renderKickBtn(); return; }
    if (document.hidden) return;
    const video = findVideo();
    if (!video) { removeKickBtn(); noBarTicks = 0; return; }
    if (!kickBtn) kickBtn = makeKickButton();

    if (kickBtn.isConnected && (scanN = (scanN + 1) % 5)) { renderKickBtn(); return; }

    const anchor = findHeaderAnchor();
    if (anchor && anchor.parentNode) {
      noBarTicks = 0;
      if (kickBtn.isConnected && kickBtnHost === 'header' && kickBtn.nextElementSibling === anchor) { renderKickBtn(); return; }
      kickBtn.classList.remove('cs-float');
      anchor.parentNode.insertBefore(kickBtn, anchor);
      kickBtnHost = 'header';
      renderKickBtn();
      return;
    }

    const bar = findControlBar(video);
    if (bar) {
      noBarTicks = 0;
      if (kickBtn.isConnected && kickBtnHost === bar && bar.contains(kickBtn)) { renderKickBtn(); return; }
      mountInBar(bar);
    } else {
      noBarTicks++;
      if (noBarTicks < 3) return;
      const root = video.closest('.video-js') || video.parentElement;
      if (!root) return;
      if (kickBtn.isConnected && kickBtnHost === root) { renderKickBtn(); return; }
      mountFloating(root);
    }
    renderKickBtn();
  }

  /* ======================= MENU DO TAMPERMONKEY ======================= */
  function registerMenu() {
    menuIds.forEach(id => { try { GM_unregisterMenuCommand(id); } catch (e) {} });
    menuIds = [];
    try {
      if (scriptEnabled) {
        menuIds.push(GM_registerMenuCommand(session ? '🍿 Desligar Cine Sharkão (voltar à live)' : '🍿 Ligar Cine Sharkão', toggleSession));
        menuIds.push(GM_registerMenuCommand('⛔ Desativar script', () => setEnabled(false)));
      } else {
        menuIds.push(GM_registerMenuCommand('✅ Ativar script', () => setEnabled(true)));
      }
    } catch (e) {}
  }

  function setEnabled(v) {
    scriptEnabled = !!v;
    GM_setValue(KEY_ENABLED, scriptEnabled);
    if (!scriptEnabled) { stopSession(); removeKickBtn(); }
    registerMenu();
    if (scriptEnabled) ensureKickButton();
  }

  function toggleSession() {
    if (!scriptEnabled) return;
    if (session) stopSession();
    else { startSession(); renderKickBtn(); registerMenu(); }
  }

  function stopSession() {
    if (!session) return;
    const s = session;
    session = null;
    try { s.cleanup(); } catch (e) {}
    renderKickBtn();
    registerMenu();
  }

      tMic.addEventListener('click', () => { lastMutedRendered = nativeVideo.muted; });

      function showControls(force) {
        const now = performance.now();
        if (!force && controlsVisible && now - lastShowAt < 400) return;
        lastShowAt = now;
        if (!controlsVisible) {
          controlsVisible = true;
          controls.style.opacity = '1';
          controls.style.pointerEvents = 'auto';
          updateTime();
          updateStatus();
        }
        clearTimeout(hideTimer);
        hideTimer = setTimeout(autoHide, 4000);
      }
      function autoHide() {
        if (menuOpen) { hideTimer = setTimeout(autoHide, 4000); return; }
        hideControls();
      }
      function hideControls() {
        closeMenu();
        controlsVisible = false;
        clearTimeout(hideTimer);
        controls.style.opacity = '0';
        controls.style.pointerEvents = 'none';
      }
      wpContainer.addEventListener('click', e => {
        const t = e.target;
        if (menuOpen && !menu.contains(t) && !gearBtn.contains(t) && !thumbWrapper.contains(t)) {
          closeMenu();
          if (!controls.contains(t)) { showControls(); return; }
        }
        if (menu.contains(t) || controls.contains(t)) { showControls(); return; }
        if (thumbWrapper.contains(t) || chatPanel.contains(t)) return;
        if (controlsVisible) hideControls(); else showControls();
      });
      controls.addEventListener('pointerdown', () => showControls());
      volSlider.addEventListener('input', () => showControls());
      wpContainer.addEventListener('keydown', () => showControls());
      showControls(true);

      if (parent) parent.appendChild(wpContainer);

      /* ---------- miniplayer ---------- */
      currentH = Math.round(currentW / cropCfg.aspect);
      const thumbWrapper = document.createElement('div');
      thumbWrapper.id = 'wp-thumb-wrapper';
      thumbWrapper.style.cssText = 'position:absolute!important;top:0!important;left:0!important;width:' + currentW + 'px!important;height:' + currentH + 'px!important;z-index:9999!important;pointer-events:auto!important;cursor:grab!important;touch-action:none!important;will-change:transform!important;border-radius:' + RADIUS + 'px!important;box-shadow:0 3px 10px rgba(0,0,0,0.55)!important;transform:translate3d(0px, 0px, 0);';

      const clipBox = document.createElement('div');
      clipBox.style.cssText = 'position:absolute!important;top:0!important;left:0!important;width:100%!important;height:100%!important;border-radius:' + RADIUS + 'px!important;overflow:hidden!important;-webkit-clip-path:inset(0 round ' + RADIUS + 'px)!important;clip-path:inset(0 round ' + RADIUS + 'px)!important;background:#000!important;isolation:isolate!important;transform:translateZ(0);';
      thumbWrapper.appendChild(clipBox);

      try {
        nativeVideo.style.cssText = getZoomCSS();
        clipBox.appendChild(nativeVideo);
      } catch (e) {}

      safeLayer.appendChild(thumbWrapper);
      renderMini();
      wpVideo.muted = false;
      setTimeout(() => { if (alive) { posX = EDGE; posY = EDGE; updateTransform(false); } }, 50);

      let lastTapTime = 0;
      thumbWrapper.addEventListener('click', e => {
        if (isEditingCrop) { e.stopPropagation(); return; }
        const now = Date.now();
        const tapLength = now - lastTapTime;
        if (tapLength < 300 && tapLength > 0) {
          isExpanded = !isExpanded;
          const screenW = vidWrapper.clientWidth, screenH = vidWrapper.clientHeight;
          currentW = isExpanded ? largeW : smallW;
          currentH = Math.round(currentW / cropCfg.aspect);
          const maxLeft = screenW - currentW - EDGE, maxTop = screenH - currentH - EDGE;
          if (posX > maxLeft) posX = Math.max(EDGE, maxLeft);
          if (posY > maxTop) posY = Math.max(EDGE, maxTop);
          updateTransform(true);
          e.stopPropagation();
        }
        lastTapTime = now;
      });

      let isDragging = false, startX = 0, startY = 0, initialPosX = 0, initialPosY = 0;
      let initialCropX = 0, initialCropY = 0;

      thumbWrapper.addEventListener('pointerdown', e => {
        isDragging = true;
        startX = e.clientX; startY = e.clientY;
        if (isEditingCrop) {
          initialCropX = cropCfg.x;
          initialCropY = cropCfg.y;
          showControls();
        } else {
          initialPosX = posX; initialPosY = posY;
          preChatPosX = null;
          thumbWrapper.style.transition = 'none';
        }
        try { thumbWrapper.setPointerCapture(e.pointerId); } catch (err) {}
        e.preventDefault();
        e.stopPropagation();
      });

      thumbWrapper.addEventListener('pointermove', e => {
        if (!isDragging) return;
        const dx = e.clientX - startX;
        const dy = e.clientY - startY;

        if (isEditingCrop) {
          // Move o conteúdo interno do vídeo baseado no arrasto
          const sensitivity = 0.6;
          cropCfg.x = Math.max(-150, Math.min(50, Math.round(initialCropX + dx * sensitivity)));
          cropCfg.y = Math.max(-150, Math.min(50, Math.round(initialCropY + dy * sensitivity)));
          slX.inp.value = cropCfg.x; slX.span.textContent = cropCfg.x + slX.unit;
          slY.inp.value = cropCfg.y; slY.span.textContent = cropCfg.y + slY.unit;
          applyCropStyle();
          return;
        }

        posX = initialPosX + dx;
        posY = initialPosY + dy;
        const maxW = vidWrapper.clientWidth, maxH = vidWrapper.clientHeight;
        posX = Math.max(EDGE, Math.min(maxW - currentW - EDGE, posX));
        posY = Math.max(EDGE, Math.min(maxH - currentH - EDGE, posY));
        if (!dragRaf) {
          dragRaf = requestAnimationFrame(() => {
            dragRaf = 0;
            thumbWrapper.style.transform = 'translate3d(' + posX + 'px, ' + posY + 'px, 0)';
          });
        }
      });

      thumbWrapper.addEventListener('pointerup', e => {
        if (!isDragging) return;
        isDragging = false;
        if (dragRaf) { cancelAnimationFrame(dragRaf); dragRaf = 0; }
        try { thumbWrapper.releasePointerCapture(e.pointerId); } catch (err) {}
        if (!isEditingCrop) snapToEdge();
      });

      thumbWrapper.addEventListener('pointercancel', () => {
        if (!isDragging) return;
        isDragging = false;
        if (dragRaf) { cancelAnimationFrame(dragRaf); dragRaf = 0; }
        if (!isEditingCrop) snapToEdge();
      });

      function loadVideo(url) {
        if (!url || url === currentUrl) return;
        lastSeekAt = -Infinity;
        const isHls = url.indexOf('.m3u8') !== -1;
        const H = getHls();
        if (isHls && H && H.isSupported()) {
          currentUrl = url;
          if (hlsInstance) { try { hlsInstance.destroy(); } catch (e) {} }
          hlsInstance = new H({ enableWorker: true, lowLatencyMode: false, maxBufferLength: 20, maxMaxBufferLength: 40, backBufferLength: 20 });
          const h = hlsInstance;
          h.loadSource(url);
          h.attachMedia(wpVideo);
          h.on(H.Events.ERROR, (ev, data) => {
            if (!data || !data.fatal) return;
            try {
              if (data.type === H.ErrorTypes.NETWORK_ERROR) h.startLoad();
              else if (data.type === H.ErrorTypes.MEDIA_ERROR) h.recoverMediaError();
              else { currentUrl = null; h.destroy(); }
            } catch (e) {}
          });
        } else if (isHls && !H && !wpVideo.canPlayType('application/vnd.apple.mpegurl')) {
          pendingUrl = url;
        } else {
          currentUrl = url;
          if (hlsInstance) { try { hlsInstance.destroy(); } catch (e) {} hlsInstance = null; }
          wpVideo.src = url;
          wpVideo.load();
        }
      }

      function seekTo(t, force) {
        const now = performance.now();
        if (!force && now - lastSeekAt < 5000) return;
        lastSeekAt = now;
        try { wpVideo.currentTime = t; } catch (e) {}
      }

      function connectSocket() {
        connectionState = 'connecting';
        socket = new WebSocket(SOCKET_URL);
        const mySocket = socket;
        mySocket.onopen = () => {
          connectionState = 'connected';
          mySocket.send(JSON.stringify({ type: 'JOIN', roomId: ROOM_ID }));
        };
        mySocket.onmessage = e => {
          try {
            const msg = JSON.parse(e.data);
            const p = msg.payload || msg;
            if (p.videoUrl) loadVideo(p.videoUrl);
            const action = p.action || msg.action;
            const serverTime = p.time !== undefined ? parseFloat(p.time) : null;
            if (serverTime !== null && !isNaN(serverTime)) {
              lastServerTime = serverTime;
              lastServerTimeReceivedAt = performance.now();
            }
            const targetTime = (serverTime !== null && !isNaN(serverTime)) ? Math.max(0, serverTime - viewerDelay) : null;
            if (action || targetTime !== null) {
              if (action === 'PLAY') {
                isPlayingFromSocket = true;
                if (targetTime !== null && !wpVideo.seeking && Math.abs(wpVideo.currentTime - targetTime) > 3) seekTo(targetTime, false);
                wpVideo.play().catch(() => {});
              } else if (action === 'PAUSE') {
                isPlayingFromSocket = false;
                wpVideo.pause();
                if (targetTime !== null && !wpVideo.seeking && Math.abs(wpVideo.currentTime - targetTime) > 2) seekTo(targetTime, true);
              } else if (targetTime !== null && !wpVideo.seeking && Math.abs(wpVideo.currentTime - targetTime) > 3) {
                seekTo(targetTime, false);
              }
            }
          } catch (err) {}
        };
        mySocket.onclose = () => {
          if (socket !== mySocket) return;
          connectionState = 'reconnecting';
          setTimeout(() => { if (alive && socket === mySocket) connectSocket(); }, 3000);
        };
        mySocket.onerror = () => { try { mySocket.close(); } catch (e) {} };
      }

      showControls(true);
      wpVideo.muted = false;
      renderVol();
      applyLayout();
      wpVideo.play().catch(() => {});
      connectSocket();

      if (!getHls()) {
        try {
          const s = document.createElement('script');
          s.src = 'https://cdn.jsdelivr.net/npm/hls.js@1.5.17/dist/hls.min.js';
          s.onload = () => { if (pendingUrl) { const u = pendingUrl; pendingUrl = null; loadVideo(u); } };
          document.head.appendChild(s);
        } catch (e) {}
      }

      function cleanup() {
        alive = false;
        timers.forEach(clearInterval);
        if (resizeObs) { try { resizeObs.disconnect(); } catch (e) {} resizeObs = null; }
        clearTimeout(hideTimer);
        if (dragRaf) { try { cancelAnimationFrame(dragRaf); } catch (e) {} dragRaf = 0; }
        if (gRaf) { try { cancelAnimationFrame(gRaf); } catch (e) {} gRaf = 0; }
        document.removeEventListener('fullscreenchange', updateFullscreenState);
        document.removeEventListener('webkitfullscreenchange', updateFullscreenState);
        if (vvHandler && window.visualViewport) {
          window.visualViewport.removeEventListener('resize', vvHandler);
          window.visualViewport.removeEventListener('scroll', vvHandler);
        }
        try {
          if (document.fullscreenElement === wpContainer) document.exitFullscreen().catch(() => {});
          else if (document.webkitFullscreenElement === wpContainer && document.webkitExitFullscreen) document.webkitExitFullscreen();
          if (screen.orientation && screen.orientation.unlock) screen.orientation.unlock();
        } catch (e) {}
        restoreKickPage();
        if (socket) {
          socket.onclose = null; socket.onerror = null; socket.onmessage = null;
          try { socket.close(); } catch (e) {}
          socket = null;
        }
        if (hlsInstance) { try { hlsInstance.destroy(); } catch (e) {} hlsInstance = null; }
        try { chatIframe.src = 'about:blank'; } catch (e) {}
        try { wpVideo.pause(); wpVideo.removeAttribute('src'); wpVideo.load(); } catch (e) {}
        try {
          if (parent) parent.insertBefore(nativeVideo, (origNext && origNext.parentNode === parent) ? origNext : null);
          if (origVideoStyle === null) nativeVideo.removeAttribute('style'); else nativeVideo.setAttribute('style', origVideoStyle);
          if (parent) {
            if (origParentStyle === null) parent.removeAttribute('style'); else parent.setAttribute('style', origParentStyle);
          }
        } catch (e) {}
        wpContainer.remove();
        try {
          if (metaCreated) meta.remove();
          else if (origMetaContent === null) meta.removeAttribute('content');
          else meta.setAttribute('content', origMetaContent);
        } catch (e) {}
        if (wakeLock) { try { wakeLock.release(); } catch (e) {} wakeLock = null; }
      }

      session = { nativeVideo, parent, path: location.pathname, cleanup };
    } catch (err) {
      alert('Cine Sharkão - erro: ' + (err && err.message ? err.message : err));
    }
  }

  function tick() {
    if (!scriptEnabled) { removeKickBtn(); return; }
    if (session && (!document.contains(session.parent) || location.pathname !== session.path)) stopSession();
    ensureKickButton();
  }

  registerMenu();
  setInterval(tick, 1000);
  tick();
})();
